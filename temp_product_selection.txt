                        const existingCells = [];
                        while (row.cells.length > 2) {
                            const cell = row.cells[2];
                            existingCells.push(cell.cloneNode(true));
                            row.deleteCell(2);
                        }
                        
                        // Add unit columns and recalculate
                        recalculateMaxColumns();

                        existingCells.forEach((cell, index) => {

                            newCell.innerHTML = cell.innerHTML;
                            newCell.className = cell.className;
                            // Re-attach event listeners
                            const input = newCell.querySelector('input');
                            if (input && cell.querySelector('input')) {
                                const oldInput = cell.querySelector('input');
                                Array.from(oldInput.attributes).forEach(attr => {
                                    input.setAttribute(attr.name, attr.value);
                                });
                            }
                        });
                        
                        row.dataset.uomColumnsAdded = 'true';
